DROP TABLE IF EXISTS orders CASCADE;
DROP TABLE IF EXISTS employees CASCADE;



CREATE TABLE employees (
    id SERIAL PRIMARY KEY,
    name VARCHAR(50),
    department VARCHAR(30),
    salary INTEGER,
    hire_date DATE
);



CREATE TABLE orders (
    id SERIAL PRIMARY KEY,
    employee_id INTEGER REFERENCES employees(id),
    product VARCHAR(50),
    amount INTEGER,
    order_date DATE
);
