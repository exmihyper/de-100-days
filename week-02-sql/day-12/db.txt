DROP TABLE IF EXISTS big_sales CASCADE;



CREATE TABLE big_sales (
    id SERIAL PRIMARY KEY,
    product VARCHAR(50),
    region VARCHAR(30),
    amount INTEGER,
    sale_date DATE
);
